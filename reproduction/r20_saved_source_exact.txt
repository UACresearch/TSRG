def sha256_file(
    path
):

    path = Path(
        path
    )

    h = hashlib.sha256()

    with path.open(
        "rb"
    ) as f:

        while True:

            chunk = f.read(
                8 * 1024 * 1024
            )

            if not chunk:
                break

            h.update(
                chunk
            )

    return h.hexdigest()


def sha256_text(
    text
):

    return hashlib.sha256(
        str(
            text
        ).encode(
            "utf-8"
        )
    ).hexdigest()


def dataset_download_fresh(
    handle
):

    fn = (
        kagglehub
        .dataset_download
    )

    sig = inspect.signature(
        fn
    )

    kwargs = {}

    if (
        "force_download"
        in sig.parameters
    ):

        kwargs[
            "force_download"
        ] = True

    return Path(
        fn(
            handle,
            **kwargs
        )
    )


def find_file(
    root,
    filename
):

    root = Path(
        root
    )

    direct = (
        root
        /
        filename
    )

    if direct.is_file():

        return direct

    hits = list(
        root.rglob(
            filename
        )
    )

    assert hits, (
        f"Could not find {filename} "
        f"under {root}"
    )

    if len(
        hits
    ) == 1:

        return hits[0]

    exact = [
        p
        for p in hits
        if p.name
        ==
        filename
    ]

    assert exact, (
        f"Ambiguous search for {filename}"
    )

    return exact[0]


def remote_candidate(
    root,
    relative_path
):

    root = Path(
        root
    )

    relative_path = Path(
        relative_path
    )

    direct = (
        root
        /
        relative_path
    )

    if direct.is_file():

        return direct

    suffix = str(
        relative_path
    ).replace(
        "\\",
        "/"
    )

    hits = []

    for p in root.rglob(
        relative_path.name
    ):

        normalized = str(
            p
        ).replace(
            "\\",
            "/"
        )

        if (
            p.is_file()
            and
            normalized.endswith(
                suffix
            )
        ):

            hits.append(
                p
            )

    if len(
        hits
    ) == 1:

        return hits[0]

    return None

def verify_remote_hashes(
    hash_map,
    max_tries=20,
    sleep_seconds=10,
):

    last_error = None

    for attempt in range(
        1,
        max_tries + 1,
    ):

        try:

            root = (
                dataset_download_fresh(
                    SAVE_HANDLE
                )
            )

            okay = True

            for (
                rel,
                expected_sha,
            ) in hash_map.items():

                p = remote_candidate(
                    root,
                    rel
                )

                if p is None:

                    okay = False

                    last_error = (
                        "missing remote file: "
                        +
                        rel
                    )

                    break

                actual_sha = (
                    sha256_file(
                        p
                    )
                )

                if (
                    actual_sha
                    !=
                    expected_sha
                ):

                    okay = False

                    last_error = (
                        "SHA mismatch: "
                        +
                        rel
                    )

                    break

            if okay:

                for rel in hash_map:

                    print(
                        "REMOTE_SHA256_VERIFIED:",
                        rel
                    )

                return True

        except Exception as exc:

            last_error = repr(
                exc
            )

        print(
            "Waiting for R20 Dataset "
            f"{attempt}/{max_tries} | "
            f"{last_error}"
        )

        time.sleep(
            sleep_seconds
        )

    raise RuntimeError(
        "Remote verification failed: "
        +
        str(
            last_error
        )
    )


def verify_manifest_dataset(
    root,
    label
):

    manifest_path = find_file(
        root,
        "sha256_manifest.json"
    )

    manifest = json.loads(
        manifest_path.read_text(
            encoding="utf-8"
        )
    )

    for (
        filename,
        expected_sha,
    ) in manifest.items():

        p = find_file(
            root,
            filename
        )

        actual_sha = (
            sha256_file(
                p
            )
        )

        assert (
            actual_sha
            ==
            expected_sha
        ), (
            f"{label} SHA mismatch: "
            f"{filename}"
        )

    print(
        f"{label}_SHA256_MANIFEST_VERIFIED: PASSED"
    )

    return manifest

def resolve_test_image(
    row
):

    image_id = str(
        row[
            "id"
        ]
    )

    category = str(
        row[
            "category"
        ]
    )

    local_path = str(
        row[
            "local_path"
        ]
    )


    candidates = []


    # Candidate 1: manifest relative path.
    if (
        local_path
        and
        local_path.lower()
        !=
        "nan"
    ):

        normalized = local_path.replace(
            "\\",
            "/"
        )

        candidates.append(
            DATA_ROOT
            /
            normalized
        )


    # Candidate 2: known locked dataset convention.
    candidates.append(
        DATA_ROOT
        /
        "images"
        /
        "test"
        /
        category
        /
        f"{image_id}.jpg"
    )


    valid = []

    for p in candidates:

        if (
            p.is_file()
            and
            p not in valid
        ):

            valid.append(
                p
            )


    if len(
        valid
    ) == 1:

        return valid[0]


    if len(
        valid
    ) > 1:

        # They should resolve to the same file bytes.
        shas = {
            sha256_file(
                p
            )
            for p in valid
        }

        assert (
            len(
                shas
            )
            ==
            1
        )

        return valid[0]


    # Last-resort exact basename search.
    hits = list(
        DATA_ROOT.rglob(
            f"{image_id}.jpg"
        )
    )


    assert (
        len(
            hits
        )
        ==
        1
    ), (
        f"Could not uniquely resolve "
        f"test image {image_id}: "
        f"{hits}"
    )


    return hits[0]

@torch.inference_mode()
def extract_radio_tokens(
    image_path
):

    with Image.open(
        image_path
    ) as src:

        image = src.convert(
            "RGB"
        )


    w, h = image.size


    side = min(
        w,
        h
    )


    left = (
        w
        -
        side
    ) // 2


    top = (
        h
        -
        side
    ) // 2


    image = image.crop(
        (
            left,
            top,
            left + side,
            top + side,
        )
    )


    image = image.resize(
        (
            512,
            512,
        ),
        Image.Resampling.BICUBIC,
    )


    x = (
        torch
        .from_numpy(
            np.asarray(
                image,
                dtype=np.uint8,
            )
            .copy()
        )
        .permute(
            2,
            0,
            1,
        )
        .float()
    )


    x = (
        2.0
        *
        x
        /
        255.0
        -
        1.0
    ).unsqueeze(
        0
    ).to(
        VIS_DEV,
        dtype=torch.bfloat16,
    )


    _, spatial = radio(
        (
            x
            +
            1.0
        )
        /
        2.0
    )


    assert (
        tuple(
            spatial.shape
        )
        ==
        (
            1,
            1024,
            1280,
        )
    )


    tokens = (
        spatial
        .reshape(
            1,
            16,
            2,
            16,
            2,
            1280,
        )
        .permute(
            0,
            1,
            3,
            2,
            4,
            5,
        )
        .reshape(
            256,
            5120,
        )
    )


    return (
        tokens
        .float()
        .cpu()
        .numpy()
    )

@torch.inference_mode()
def generate_from_cache(
    language_model,
    cache_array,
    positions,
):

    visual_np = np.stack(
        [
            np.asarray(
                cache_array[
                    int(
                        pos
                    )
                ],
                dtype=np.float32,
            )
            for pos in positions
        ],
        axis=0,
    )


    visual = (
        torch
        .from_numpy(
            visual_np
        )
        .to(
            VIS_DEV
        )
    )


    image_embeds = (
        projector(
            visual
        )
        .to(
            LLM_DEV,
            dtype=DTYPE,
        )
    )


    del visual
    del visual_np


    bsz = len(
        positions
    )


    prefix = (
        PREFIX_IDS_T
        .repeat(
            bsz,
            1,
        )
    )


    suffix = (
        SUFFIX_IDS_T
        .repeat(
            bsz,
            1,
        )
    )


    prefix_embeds = (
        language_model
        .get_input_embeddings()(
            prefix
        )
    )


    suffix_embeds = (
        language_model
        .get_input_embeddings()(
            suffix
        )
    )


    inputs_embeds = torch.cat(
        [
            prefix_embeds,
            image_embeds,
            suffix_embeds,
        ],
        dim=1,
    )


    seq_len = (
        inputs_embeds.shape[
            1
        ]
    )


    attention_mask = torch.ones(
        (
            bsz,
            seq_len,
        ),
        dtype=torch.long,
        device=LLM_DEV,
    )


    position_ids = (
        torch.arange(
            seq_len,
            device=LLM_DEV,
            dtype=torch.long,
        )
        .unsqueeze(
            0
        )
        .repeat(
            bsz,
            1,
        )
    )


    generated = []

    past = None


    finished = torch.zeros(
        bsz,
        dtype=torch.bool,
        device=LLM_DEV,
    )


    for _ in range(
        MAX_NEW_TOKENS
    ):

        output = language_model(
            inputs_embeds=inputs_embeds,
            attention_mask=attention_mask,
            position_ids=position_ids,
            past_key_values=past,
            use_cache=True,
            return_dict=True,
        )


        next_ids = torch.argmax(
            output.logits[
                :,
                -1,
                :
            ],
            dim=-1,
            keepdim=True,
        )


        if (
            tok.eos_token_id
            is not None
        ):

            eos_fill = torch.full_like(
                next_ids,
                tok.eos_token_id,
            )


            next_ids = torch.where(
                finished[
                    :,
                    None
                ],
                eos_fill,
                next_ids,
            )


            finished = (
                finished
                |
                (
                    next_ids[
                        :,
                        0
                    ]
                    ==
                    tok.eos_token_id
                )
            )


        generated.append(
            next_ids
        )


        past = (
            output
            .past_key_values
        )


        if bool(
            finished.all()
        ):

            break


        inputs_embeds = (
            language_model
            .get_input_embeddings()(
                next_ids
            )
        )


        attention_mask = torch.cat(
            [
                attention_mask,
                torch.ones(
                    (
                        bsz,
                        1,
                    ),
                    dtype=torch.long,
                    device=LLM_DEV,
                ),
            ],
            dim=1,
        )


        position_ids = (
            position_ids[
                :,
                -1:
            ]
            +
            1
        )


    generated_ids = torch.cat(
        generated,
        dim=1,
    )


    texts = tok.batch_decode(
        generated_ids.cpu(),
        skip_special_tokens=True,
    )


    return [
        t.strip()
        for t in texts
    ]


# ==========================================================================================
# 20. CAMERA PARSER
# ==========================================================================================

NUMBER = (
    r"[+-]?"
    r"(?:\d+(?:\.\d+)?|\.\d+)"
    r"(?:[eE][+-]?\d+)?"
)


CAMERA_RE = re.compile(
    (
        rf"camera parameters"
        rf".{{0,100}}?"
        rf"are\s*:\s*"
        rf"({NUMBER})\s*,\s*"
        rf"({NUMBER})\s*,\s*"
        rf"({NUMBER})"
    ),
    re.I,
)


def parse_camera(
    text
):

    matches = list(
        CAMERA_RE.finditer(
            str(
                text
            )
        )
    )


    if not matches:

        return None


    m = matches[
        -1
    ]


    rad = [
        float(
            m.group(
                i
            )
        )
        for i in [
            1,
            2,
            3,
        ]
    ]


    return tuple(
        math.degrees(
            x
        )
        for x in rad
    )

    .agg(

        n=(

    cluster[
        "n"
    ]

        dtype=float
    )

    cluster[
        "sum_m0"
    ]
    .to_numpy(
        dtype=float
    )
)


cluster_m2 = (
    cluster[
        "sum_m2"
    ]
    .to_numpy(
        dtype=float
    )
)


K = len(
    cluster
)


rng = np.random.default_rng(
    BOOTSTRAP_SEED
)


boot_delta = np.empty(
    BOOTSTRAP_B,
    dtype=float,
)


for b in range(
    BOOTSTRAP_B
):

    sampled = rng.integers(
        low=0,
        high=K,
        size=K,
    )


    N = (
        cluster_n[
            sampled
        ]
        .sum()
    )


    m0_b = (
        cluster_m0[
            sampled
        ]
        .sum()
        /
        N
    )


    m2_b = (
        cluster_m2[
            sampled
        ]
        .sum()
        /
        N
    )


    boot_delta[
        b
    ] = (
        m2_b
        -
        m0_b
    )

    boot_delta,

        2.5,
        97.5,

)
